Example: Fetch Action State Handler

[Event] -> fetch -> [Target]

data-flow="fetch -> #output-13"

data-flow="fetch -> #output-13" is an abreviation for data-flow="click -> fetch -> #output-13"

<button
  data-flow="fetch -> #output-13"
  data-method="get"
  data-url='data:application/json;charset=utf-8,"Hello World!"'>Usual behaviour</button>
  
<button
  data-flow="fetch -> #output-13"
  data-method="get"
  data-on-success="fetchCustomHandler.on.success"
  data-url='data:application/json;charset=utf-8,{"success": true, "html":"Hello Success!"}'>Success Data</button>
  
<button
  data-flow="fetch -> #output-13"
  data-method="get"
  data-on-fail="fetchCustomHandler.on.fail"
  data-url='data:application/json;charset=utf-8,{"success": false, "html":"Hello Failure!"}'>Fail Data</button>
  
<button
  data-flow="fetch -> #output-13"
  data-method="get"
  data-on-error="fetchCustomHandler.on.error"
  data-url="/invalid-url-404">Error: 404</button>
  
<button 
  data-flow="fetchCustomHandler.clear -> .output-block">Clear</button>

<div class="output flex-output">
  <span>Target:</span>
  <div id="output-13" class="output-block"><em>Output will appear here...</em></div>
  <span>Success:</span>
  <div id="output-13-success" class="output-block"><em>Success Output will appear here...</em></div>
  <span>Failure:</span>
  <div id="output-13-fail" class="output-block"><em>Fail Output will appear here...</em></div>
  <span>Errror:</span>
  <div id="output-13-error" class="output-block"><em>Error Output will appear here...</em></div>
</div>

<script>
  const flow = new AviFlow({
    fetchCustomHandler: {
    clear: () => {
      return 'Output will appear here...';
    },
    on: {
      error: (data, element, event, targetSelector, flow) => {
        const target = document.getElementById('output-13-error');
        flow.tools.element.setContent(target, `The error is: ${data}` );
      },
      fail: (data, element, event, targetSelector, flow) => {
        const target = document.getElementById('output-13-fail');
        flow.tools.element.setContent(target, `Failure: ${data.html}`);
      },
      success: (data, element, event, targetSelector, flow) => {
        const target = document.getElementById('output-13-success');
        flow.tools.element.setContent(target, `Success: ${data.html}`);
      }
    }
  },

  });
</script>
